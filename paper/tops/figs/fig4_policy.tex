% Fig. 5 -- T3 policy benchmark (allocation objectives vs. profile error).
% Series identified by a frameless legend (top-right); the only in-plot
% annotation is the 3.5x objective-mismatch bracket at n=400.
% Data: figs/figdata/fig4_policy.dat (cols: n uni ney ucb psi).
\begin{tikzpicture}
\begin{axis}[
  tomog,
  width=\columnwidth, height=6.8cm,
  xlabel={probe budget $n$},
  ylabel={mean profile error},
  xmode=log,
  xtick={25,50,100,200,400},
  xticklabels={25,50,100,200,400},
  log ticks with fixed point,
  xmin=21, xmax=1050,
  yticklabel style={/pgf/number format/.cd, fixed, precision=2},
  ymin=0.02, ymax=0.19,
  xmajorgrids=false, ymajorgrids=true,
  legend style={at={(0.985,0.985)}, anchor=north east},
]
  \addplot[black, thick, mark=*, mark size=1.5pt]
    table[x=n, y=uni] {figs/figdata/fig4_policy.dat};
  \addlegendentry{uniform}
  \addplot[oiblue, thick, mark=square*, mark size=1.4pt]
    table[x=n, y=ney] {figs/figdata/fig4_policy.dat};
  \addlegendentry{Neyman (variance-equal)}
  \addplot[oired, semithick, densely dashed, mark=triangle*, mark size=1.6pt]
    table[x=n, y=ucb] {figs/figdata/fig4_policy.dat};
  \addlegendentry{UCB-lite (regret)}
  \addplot[oiorange, semithick, densely dashed, mark=*, mark size=1.5pt]
    table[x=n, y=psi] {figs/figdata/fig4_policy.dat};
  \addlegendentry{PSI (information)}
  % endpoint value labels (reference-style direct labeling)
  \node[font=\tiny, text=oiorange, anchor=south, inner sep=2pt]
    at (axis cs:400,0.129) {$0.129$};
  \node[font=\tiny, text=oired, anchor=west, inner sep=3pt]
    at (axis cs:415,0.121) {$0.121$};
  \node[font=\tiny, text=black!70, anchor=north, inner sep=2pt]
    at (axis cs:400,0.035) {$0.035$};
  % the objective-mismatch gap at n=400 (uniform 0.035 vs UCB 0.121),
  % anchored on the curves' own endpoints
  \draw[oired, <->] (axis cs:400,0.035) -- (axis cs:400,0.121)
    node[pos=0.5, right=3pt, font=\scriptsize, text=oired, inner sep=1pt]
    {3.5$\times$};
\end{axis}
\end{tikzpicture}
